import { factories } from '@strapi/strapi';

export default factories.createCoreController('api::order.order', ({ strapi }) => ({

  // POST /custom-functions/getPublicMenu  { businessId, tableId }
  async getPublicMenu(ctx) {
    try {
      const { businessId, tableId } = ctx.request.body;
      if (!businessId) return ctx.badRequest('businessId is required');

      const business = await strapi.db.query('api::business.business').findOne({
        where: { id: businessId },
        populate: ['logo'],
      });
      if (!business) return ctx.notFound('Business not found');

      const table = tableId
        ? await strapi.db.query('api::table.table').findOne({ where: { id: tableId } })
        : null;

      const categories = await strapi.db.query('api::menu-category.menu-category').findMany({
        where: { business: businessId, is_active: true },
        orderBy: { sort_order: 'asc' },
      });

      const items = await strapi.db.query('api::menu-item.menu-item').findMany({
        where: { business: businessId },
        populate: ['image', 'menu_category'],
      });

      ctx.send({
        business: {
          id: business.id,
          business_name: business.business_name,
          business_type: business.business_type,
          plan_type: business.plan_type,
          currency: business.currency,
          city: business.city,
          country: business.country,
          logo: business.logo?.url || null,
        },
        table: table ? {
          id: table.id,
          table_number: table.table_number,
          branch_id: table.branch?.id || null,
          assigned_waiter_id: table.assigned_waiter?.id || null,
          assigned_waiter_name: table.assigned_waiter?.full_name || null,
        } : null,
        categories: categories.map(c => ({ id: c.id, name: c.name, icon: c.icon, sort_order: c.sort_order })),
        items: items.map(i => ({
          id: i.id,
          name: i.name,
          description: i.description,
          price: i.price,
          image: i.image?.url || null,
          is_available: i.is_available,
          is_popular: i.is_popular,
          is_sponsored: i.is_sponsored,
          preparation_time: i.preparation_time,
          tags: i.tags,
          category_id: i.menu_category?.id || null,
        })),
      });
    } catch (err) {
      ctx.throw(500, err.message);
    }
  },

  // POST /custom-functions/getClientOrders  { customerSessionId }
  async getClientOrders(ctx) {
    try {
      const { customerSessionId } = ctx.request.body;
      if (!customerSessionId) return ctx.send({ orders: [] });

      const orders = await strapi.db.query('api::order.order').findMany({
        where: { customer_session_id: customerSessionId },
        orderBy: { createdAt: 'desc' },
        limit: 50,
      });

      ctx.send({
        orders: orders.map(o => ({
          id: o.id,
          order_number: o.order_number,
          status: o.status,
          payment_status: o.payment_status,
          items: o.items,
          subtotal: o.subtotal,
          service_charge: o.service_charge,
          total: o.total,
          notes: o.notes,
          created_date: o.createdAt,
        })),
      });
    } catch (err) {
      ctx.throw(500, err.message);
    }
  },

  // POST /custom-functions/placeOrder
  // { businessId, branchId, tableId, tableNumber, waiterId, waiterName, customerSessionId, items, notes }
  async placeOrder(ctx) {
    try {
      const {
        businessId, tableId, tableNumber, waiterId,
        customerSessionId, items = [], notes = '',
      } = ctx.request.body;

      if (!businessId || !tableId || !items.length) {
        return ctx.badRequest('businessId, tableId and items are required');
      }

      const subtotal = items.reduce((s, i) => s + (i.price * i.quantity), 0);
      const business = await strapi.db.query('api::business.business').findOne({ where: { id: businessId } });
      const serviceChargePercent = business?.service_charge_percent || 0;
      const serviceCharge = +(subtotal * (serviceChargePercent / 100)).toFixed(2);
      const total = +(subtotal + serviceCharge).toFixed(2);

      const orderNumber = `ORD-${Date.now().toString(36).toUpperCase()}`;

      const order = await strapi.db.query('api::order.order').create({
        data: {
          order_number: orderNumber,
          status: 'pending',
          payment_status: 'unpaid',
          subtotal,
          service_charge: serviceCharge,
          total,
          notes,
          customer_session_id: customerSessionId,
          items: items.map(i => ({
            name: i.name,
            quantity: i.quantity,
            price: i.price,
            notes: i.notes || '',
            status: 'pending',
          })),
          table: tableId,
          business: businessId,
          waiter: waiterId || null,
          publishedAt: new Date(),
        },
      });

      // Update table status
      await strapi.db.query('api::table.table').update({
        where: { id: tableId },
        data: { status: 'ordering' },
      });

      ctx.send({ success: true, order: { id: order.id, order_number: order.order_number } });
    } catch (err) {
      ctx.throw(500, err.message);
    }
  },

}));

import { factories } from '@strapi/strapi';

export default factories.createCoreController('api::ad-campaign.ad-campaign', ({ strapi }) => ({

  // POST /custom-functions/getAdCampaigns
  // { placementKey, businessId, planType, city, country, businessType }
  async getAdCampaigns(ctx) {
    try {
      const { placementKey, businessId, planType, city, country, businessType } = ctx.request.body;
      const today = new Date().toISOString().slice(0, 10);

      let adSettings = null;
      if (businessId) {
        adSettings = await strapi.db.query('api::business-ad-setting.business-ad-setting').findOne({
          where: { business: businessId },
        });
      }

      const platform = await strapi.db.query('api::ad-campaign.ad-campaign').findMany({
        where: { campaign_type: 'platform_direct', status: 'active', placement: placementKey },
        populate: ['image_url', 'video_url', 'mobile_image_url', 'mobile_video_url', 'tablet_image_url', 'tablet_video_url', 'desktop_image_url', 'desktop_video_url'],
        orderBy: { priority: 'asc' },
      });

      const business = businessId
        ? await strapi.db.query('api::ad-campaign.ad-campaign').findMany({
            where: { campaign_type: 'business_own', status: 'active', placement: placementKey, business: businessId },
            populate: ['image_url', 'video_url', 'mobile_image_url', 'mobile_video_url', 'tablet_image_url', 'tablet_video_url', 'desktop_image_url', 'desktop_video_url'],
            orderBy: { priority: 'asc' },
          })
        : [];

      const isActive = (c) => {
        if (c.start_date && c.start_date > today) return false;
        if (c.end_date && c.end_date < today) return false;
        if (c.target_plan_types?.length && planType && !c.target_plan_types.includes(planType)) return false;
        if (c.target_countries?.length && country && !c.target_countries.includes(country)) return false;
        if (c.target_cities?.length && city && !c.target_cities.includes(city)) return false;
        if (c.target_business_types?.length && businessType && !c.target_business_types.includes(businessType)) return false;
        return true;
      };

      let chosen = [];
      let mode = 'platform_ads';
      const rejected = [];

      if (adSettings?.platform_admin_override) {
        mode = adSettings.ad_mode;
        if (mode === 'no_ads') chosen = [];
        else if (mode === 'platform_ads') chosen = platform.filter(isActive);
        else if (mode === 'business_ads_only') chosen = business.filter(isActive);
        else if (mode === 'mixed_ads') chosen = [...platform, ...business].filter(isActive);
      } else if (adSettings?.force_platform_ads) {
        chosen = platform.filter(isActive);
      } else if (!planType || planType === 'basic') {
        chosen = platform.filter(isActive);
        if (chosen.length === 0) chosen = business.filter(isActive);
      } else {
        mode = adSettings?.ad_mode || 'no_ads';
        if (mode === 'business_ads_only') chosen = business.filter(isActive);
        else if (mode === 'platform_ads') chosen = platform.filter(isActive);
        else if (mode === 'mixed_ads') chosen = [...platform, ...business].filter(isActive);
      }

      const mediaUrl = (m) => m?.url || null;
      const mapped = chosen.map(c => ({
        id: c.id,
        campaign_name: c.campaign_name,
        title: c.title,
        description: c.description,
        cta_text: c.cta_text,
        cta_link: c.cta_link,
        image_url: mediaUrl(c.image_url),
        video_url: mediaUrl(c.video_url),
        mobile_image_url: mediaUrl(c.mobile_image_url),
        mobile_video_url: mediaUrl(c.mobile_video_url),
        tablet_image_url: mediaUrl(c.tablet_image_url),
        tablet_video_url: mediaUrl(c.tablet_video_url),
        desktop_image_url: mediaUrl(c.desktop_image_url),
        desktop_video_url: mediaUrl(c.desktop_video_url),
        impressions: c.impressions,
        clicks: c.clicks,
      }));

      ctx.send({
        campaigns: mapped,
        useAdSense: false,
        debug: { adMode: mode, total_fetched: platform.length + business.length, total_matched: mapped.length, rejected },
      });
    } catch (err) {
      ctx.throw(500, err.message);
    }
  },

}));

import { factories } from '@strapi/strapi';
import crypto from 'crypto';

const INVITE_BASE = process.env.FRONTEND_URL || 'https://juicy-dine-smart-flow.base44.app';

export default factories.createCoreController('api::employee-invitation.employee-invitation', ({ strapi }) => ({

  // POST /custom-functions/sendEmployeeInvite
  // { invited_name, invited_email, phone, role, branch_id, business_id, business_name }
  async sendEmployeeInvite(ctx) {
    try {
      const { invited_name, invited_email, phone, role, branch_id, business_id, business_name } = ctx.request.body;
      if (!invited_name || !invited_email || !role || !business_id) {
        return ctx.send({ error: 'invited_name, invited_email, role and business_id are required' }, 400);
      }

      // Cancel any prior pending invite for the same email + business
      const existing = await strapi.db.query('api::employee-invitation.employee-invitation').findMany({
        where: { invited_email, business: business_id, status: 'pending' },
      });
      for (const inv of existing) {
        await strapi.db.query('api::employee-invitation.employee-invitation').update({
          where: { id: inv.id },
          data: { status: 'cancelled' },
        });
      }

      const token = crypto.randomBytes(24).toString('hex');
      const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000); // 7 days

      const invitation = await strapi.db.query('api::employee-invitation.employee-invitation').create({
        data: {
          invited_name,
          invited_email,
          phone: phone || '',
          role,
          branch: branch_id || null,
          business: business_id,
          invite_token: token,
          status: 'pending',
          expires_at: expiresAt,
          publishedAt: new Date(),
        },
      });

      const inviteLink = `${INVITE_BASE}/accept-invite?token=${token}`;

      let emailSent = false;
      try {
        await strapi.plugins['email'].services.email.send({
          to: invited_email,
          subject: `You've been invited to join ${business_name || 'a restaurant'} on SmartMenu`,
          text: `Hi ${invited_name},\n\nYou've been invited to join ${business_name || 'a restaurant'} as a ${role}.\n\nAccept your invite here: ${inviteLink}\n\nThis link expires in 7 days.`,
        });
        emailSent = true;
      } catch (mailErr) {
        strapi.log.warn(`[sendEmployeeInvite] Email send failed: ${mailErr.message}`);
        emailSent = false;
      }

      ctx.send({
        success: true,
        invitationId: invitation.id,
        inviteLink,
        emailSent,
      });
    } catch (err) {
      ctx.throw(500, err.message);
    }
  },

  // POST /custom-functions/validateInviteToken  { token }
  async validateInviteToken(ctx) {
    try {
      const { token } = ctx.request.body;
      if (!token) return ctx.send({ valid: false, reason: 'invalid' });

      const invitation = await strapi.db.query('api::employee-invitation.employee-invitation').findOne({
        where: { invite_token: token },
        populate: ['business', 'branch'],
      });

      if (!invitation) return ctx.send({ valid: false, reason: 'invalid' });
      if (invitation.status === 'accepted') return ctx.send({ valid: false, reason: 'accepted' });
      if (invitation.status === 'cancelled') return ctx.send({ valid: false, reason: 'invalid' });
      if (invitation.expires_at && new Date(invitation.expires_at) < new Date()) {
        if (invitation.status !== 'expired') {
          await strapi.db.query('api::employee-invitation.employee-invitation').update({
            where: { id: invitation.id },
            data: { status: 'expired' },
          });
        }
        return ctx.send({ valid: false, reason: 'expired' });
      }

      ctx.send({
        valid: true,
        invitation: {
          id: invitation.id,
          invited_name: invitation.invited_name,
          invited_email: invitation.invited_email,
          role: invitation.role,
        },
        business: invitation.business ? {
          id: invitation.business.id,
          business_name: invitation.business.business_name,
        } : null,
        branch: invitation.branch ? {
          id: invitation.branch.id,
          branch_name: invitation.branch.branch_name,
        } : null,
      });
    } catch (err) {
      ctx.throw(500, err.message);
    }
  },

  // POST /custom-functions/acceptInvite  { token }  — requires authenticated user (ctx.state.user)
  async acceptInvite(ctx) {
    try {
      const { token } = ctx.request.body;
      const user = ctx.state.user;
      if (!user) return ctx.send({ success: false, error: 'Not authenticated' }, 401);
      if (!token) return ctx.send({ success: false, error: 'Token is required' }, 400);

      const invitation = await strapi.db.query('api::employee-invitation.employee-invitation').findOne({
        where: { invite_token: token },
        populate: ['business', 'branch'],
      });

      if (!invitation) return ctx.send({ success: false, error: 'Invalid invite link' });
      if (invitation.status !== 'pending') return ctx.send({ success: false, error: 'Invitation already used or expired' });
      if (invitation.invited_email.toLowerCase() !== user.email.toLowerCase()) {
        return ctx.send({ success: false, error: 'This invite was sent to a different email address' });
      }

      const DEFAULT_PERMISSIONS = {
        manager: { view_orders: true, manage_menu: true, assign_waiters: true, view_daily_reports: true, approve_cancellations: true },
        waiter: { view_assigned_tables: true, receive_orders: true, update_order_status: true, receive_waiter_calls: true },
      };

      const employee = await strapi.db.query('api::employee.employee').create({
        data: {
          full_name: invitation.invited_name,
          phone: invitation.phone || '',
          role: invitation.role,
          is_active: true,
          permissions: DEFAULT_PERMISSIONS[invitation.role] || {},
          user: user.id,
          business: invitation.business.id,
          branch: invitation.branch?.id || null,
          publishedAt: new Date(),
        },
      });

      await strapi.db.query('api::employee-invitation.employee-invitation').update({
        where: { id: invitation.id },
        data: { status: 'accepted' },
      });

      ctx.send({
        success: true,
        employeeId: employee.id,
        business_id: invitation.business.id,
        role: invitation.role,
      });
    } catch (err) {
      ctx.throw(500, err.message);
    }
  },

}));

import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';

export const DEFAULT_SETTINGS = {
  theme_style: 'modern_dark',
  primary_color: '#F59E0B',
  accent_color: '#FBBF24',
  button_color: '#F59E0B',
  text_color: '#FFFFFF',
  background_color: '#080808',
  card_color: '#141414',
  layout_style: 'classic_list',
  header_style: 'floating_glass',
  category_style: 'rounded_pills',
  product_card_style: 'large_image',
  show_hero: true,
  show_ai_recommendations: true,
  status: 'draft',
};

export function useMenuSettings(businessId) {
  const queryClient = useQueryClient();

  const { data: rawSettings, isLoading } = useQuery({
    queryKey: ['menuSettings', businessId],
    queryFn: async () => {
      if (!businessId) return null;
      const results = await base44.entities.BusinessMenuSetting.filter({ business: businessId });
      const sorted = results.sort((a, b) => new Date(b.updatedAt) - new Date(a.updatedAt));
      return sorted[0] || null;
    },
    enabled: !!businessId,
    staleTime: 0,
    refetchOnWindowFocus: false,
  });

  const getOrFetchRecord = async () => {
    const results = await base44.entities.BusinessMenuSetting.filter({ business: businessId });
    const sorted = results.sort((a, b) => new Date(b.updatedAt) - new Date(a.updatedAt));
    return sorted[0] || null;
  };

  const saveDraftMutation = useMutation({
    mutationFn: async (data) => {
      const existing = await getOrFetchRecord();
      const payload = { ...data, business: businessId, status: 'draft' };
      if (existing?.id) return base44.entities.BusinessMenuSetting.update(existing.id, payload);
      return base44.entities.BusinessMenuSetting.create(payload);
    },
    onSuccess: (result) => {
      queryClient.setQueryData(['menuSettings', businessId], result);
      queryClient.setQueryData(['publicMenuSettings', businessId], result);
    },
  });

  const publishMutation = useMutation({
    mutationFn: async (data) => {
      const existing = await getOrFetchRecord();
      const payload = {
        ...data,
        business: businessId,
        status: 'published',
        published_at: new Date().toISOString(),
      };
      if (existing?.id) return base44.entities.BusinessMenuSetting.update(existing.id, payload);
      return base44.entities.BusinessMenuSetting.create(payload);
    },
    onSuccess: (result) => {
      queryClient.setQueryData(['menuSettings', businessId], result);
      queryClient.setQueryData(['publicMenuSettings', businessId], result);
      queryClient.invalidateQueries({ queryKey: ['publicMenuSettings', businessId], refetchType: 'all' });
    },
  });

  const merged = { ...DEFAULT_SETTINGS, ...rawSettings };

  return {
    settings: merged,
    rawSettings,
    isLoading,
    saveDraft: saveDraftMutation.mutate,
    savingDraft: saveDraftMutation.isPending,
    publish: publishMutation.mutate,
    publishing: publishMutation.isPending,
    save: publishMutation.mutate,
    saving: publishMutation.isPending,
  };
}

export function usePublicMenuSettings(businessId) {
  const { data, isLoading } = useQuery({
    queryKey: ['publicMenuSettings', businessId],
    queryFn: async () => {
      if (!businessId) return null;
      const results = await base44.entities.BusinessMenuSetting.filter({ business: businessId });
      if (!results || results.length === 0) return null;
      const sorted = results.sort((a, b) => new Date(b.updatedAt) - new Date(a.updatedAt));
      const published = sorted.find(r => r.status === 'published');
      return published || sorted[0];
    },
    enabled: !!businessId,
    staleTime: 0,
    refetchOnMount: true,
    refetchOnWindowFocus: true,
  });

  // Strapi media fields come back as relation objects — flatten to plain URLs
  const flattenMedia = (raw) => {
    if (!raw) return raw;
    return {
      ...raw,
      cover_image_url: raw.cover_image_url?.url || raw.cover_image_url || null,
      cover_video_url: raw.cover_video_url?.url || raw.cover_video_url || null,
    };
  };

  const settings = { ...DEFAULT_SETTINGS, ...flattenMedia(data) };

  return { settings, isLoading, raw: data };
}

/**
 * Resolves which ad campaigns should show for a given business/placement.
 * Mirrors the logic baked into the Strapi ad-campaign.getAdCampaigns controller,
 * for places that want client-side filtering (e.g. the Ads management UI) rather
 * than the per-placement runtime hook (useAdForPlacement).
 */
export function useMenuAds(businessId, planType) {
  const { data: adSettingsArr } = useQuery({
    queryKey: ['bizAdSettings', businessId],
    queryFn: () => base44.entities.BusinessAdSetting.filter({ business: businessId }),
    enabled: !!businessId,
    staleTime: 60000,
  });

  const { data: platformCampaigns = [] } = useQuery({
    queryKey: ['activePlatformAds'],
    queryFn: () => base44.entities.AdCampaign.filter({ campaign_type: 'platform_direct', status: 'active' }),
    staleTime: 60000,
  });

  const { data: businessCampaigns = [] } = useQuery({
    queryKey: ['activeBusinessAds', businessId],
    queryFn: () => base44.entities.AdCampaign.filter({ campaign_type: 'business_own', status: 'active', business: businessId }),
    enabled: !!businessId,
    staleTime: 60000,
  });

  const settings = adSettingsArr?.[0];
  const today = new Date().toISOString().slice(0, 10);

  const isActive = (c) => {
    if (c.status !== 'active') return false;
    if (c.start_date && c.start_date > today) return false;
    if (c.end_date && c.end_date < today) return false;
    if (c.target_plan_types?.length > 0 && !c.target_plan_types.includes(planType)) return false;
    return true;
  };

  const getAdsForPlacement = (placement) => {
    const platform = platformCampaigns.filter(c => isActive(c) && c.placement === placement).sort((a, b) => (a.priority || 5) - (b.priority || 5));
    const business = businessCampaigns.filter(c => isActive(c) && c.placement === placement).sort((a, b) => (a.priority || 5) - (b.priority || 5));

    if (settings?.platform_admin_override) {
      const overrideMode = settings.override_ad_mode || settings.ad_mode;
      if (overrideMode === 'no_ads') return [];
      if (overrideMode === 'platform_ads') return platform;
      if (overrideMode === 'business_ads_only') return business;
      if (overrideMode === 'mixed_ads') return [...platform, ...business];
    }

    if (settings?.force_platform_ads) return platform;

    if (!planType || planType === 'basic') {
      if (platform.length > 0) return platform;
      if (business.length > 0) return business;
      return [];
    }

    const mode = settings?.ad_mode || 'no_ads';
    if (mode === 'no_ads') return [];
    if (mode === 'business_ads_only') return business;
    if (mode === 'platform_ads') return platform;
    if (mode === 'mixed_ads') return [...platform, ...business].sort((a, b) => (a.priority || 5) - (b.priority || 5));
    return [];
  };

  return { getAdsForPlacement, settings };
}

export function getDeviceType() {
  const w = window.innerWidth;
  if (w < 768) return 'mobile';
  if (w < 1024) return 'tablet';
  return 'desktop';
}

function resolveField(ad, ...keys) {
  for (const k of keys) {
    const v = ad[k];
    if (v && typeof v === 'string' && v.trim() !== '') return v;
  }
  return null;
}

export function getDeviceCreative(ad, forceDevice) {
  if (!ad) return { image_url: null, video_url: null, device: null };
  const device = forceDevice || getDeviceType();

  let image_url = null;
  let video_url = null;

  if (device === 'mobile') {
    image_url = resolveField(ad, 'mobile_image_url', 'image_url');
    video_url = resolveField(ad, 'mobile_video_url', 'video_url');
  } else if (device === 'tablet') {
    image_url = resolveField(ad, 'tablet_image_url', 'image_url');
    video_url = resolveField(ad, 'tablet_video_url', 'video_url');
  } else {
    image_url = resolveField(ad, 'desktop_image_url', 'image_url');
    video_url = resolveField(ad, 'desktop_video_url', 'video_url');
  }

  return { image_url, video_url, device };
}

// ── Main hook — now calls the Strapi custom-functions/getAdCampaigns route ────
export function useAdForPlacement({
  businessId,
  placementKey,
  planType,
  city,
  country,
  businessType,
  demoMode = false,
  refreshInterval = 60000,
}) {
  const [ad, setAd] = useState(null);
  const [loading, setLoading] = useState(true);
  const [debugInfo, setDebugInfo] = useState(null);
  const [useAdSense, setUseAdSense] = useState(false);

  const fetchAd = useCallback(async () => {
    if (demoMode) {
      setAd(DEMO_ADS[placementKey] || null);
      setDebugInfo({ mode: 'demo', reason: 'demo mode — local sample ad' });
      setUseAdSense(false);
      setLoading(false);
      return;
    }

    try {
      const res = await base44.functions.invoke('getAdCampaigns', {
        placementKey,
        businessId,
        planType,
        city,
        country,
        businessType,
      });

      const { campaigns = [], debug = {}, useAdSense: adsense = false } = res || {};
      setUseAdSense(adsense);

      if (campaigns.length > 0) {
        setAd(campaigns[0]);
        setDebugInfo({
          mode: 'live',
          adMode: debug.adMode,
          totalFetched: debug.total_fetched,
          totalMatched: debug.total_matched,
          rejected: debug.rejected,
          selectedCampaign: campaigns[0].campaign_name || campaigns[0].title,
          reason: 'live campaign matched',
        });
      } else {
        setAd(null);
        setDebugInfo({
          mode: 'live',
          adMode: debug.adMode,
          totalFetched: debug.total_fetched,
          totalMatched: 0,
          rejected: debug.rejected,
          reason: adsense ? 'no campaign — AdSense fallback active' : 'no active campaigns for this placement',
        });
      }
    } catch (err) {
      setAd(null);
      setDebugInfo({ mode: 'live', reason: `error: ${err.message}` });
    }

    setLoading(false);
  }, [businessId, placementKey, planType, city, country, businessType, demoMode]);

  useEffect(() => {
    setLoading(true);
    fetchAd();

    if (!demoMode && refreshInterval > 0) {
      const timer = setInterval(fetchAd, refreshInterval);
      return () => clearInterval(timer);
    }
  }, [fetchAd, demoMode, refreshInterval]);

  return { ad, loading, debugInfo, useAdSense, refresh: fetchAd };
}

// ── Deep flatten one Strapi item { id, attributes: { ... } } ─────────────────
function flattenItem(item) {
  if (!item || typeof item !== 'object') return item;
  const result = { id: item.id };
  for (const [key, value] of Object.entries(item.attributes || {})) {
    if (value === null || value === undefined) {
      result[key] = value;
      continue;
    }
    // Strapi relation: { data: null | { id, attributes } | [...] }
    if (typeof value === 'object' && !Array.isArray(value) && 'data' in value) {
      if (value.data === null) {
        result[key] = null;
        result[`${key}_id`] = null;
      } else if (Array.isArray(value.data)) {
        result[key] = value.data.map(flattenItem);
        result[`${key}_ids`] = value.data.map((i) => i.id);
      } else {
        const nested = flattenItem(value.data);
        result[key] = nested;
        result[`${key}_id`] = value.data.id;
      }
    }
    // Strapi media file object (has 'url' string directly)
    else if (typeof value === 'object' && !Array.isArray(value) && typeof value.url === 'string') {
      result[key] = value.url ? `${STRAPI_URL}${value.url}` : null;
    }
    // Repeatable component array (e.g. order items)
    else if (Array.isArray(value)) {
      result[key] = value.map((v) =>
        v && typeof v === 'object' && 'id' in v && '__component' in (v || {})
          ? v
          : typeof v === 'object' && v !== null
          ? flattenItem({ id: v.id, attributes: v })
          : v
      );
    } else {
      result[key] = value;
    }
  }
  return result;
}

// Flatten the full Strapi REST envelope { data: [...] | {...}, meta }
function flattenResponse(strapiEnvelope) {
  if (!strapiEnvelope || !strapiEnvelope.data) return strapiEnvelope;
  if (Array.isArray(strapiEnvelope.data))
    return strapiEnvelope.data.map(flattenItem);
  return flattenItem(strapiEnvelope.data);
}

// Convert flat { business_id: x, status: 'y' } → Strapi filter format
function buildStrapiFilters(params) {
  const result = {};
  for (const [key, value] of Object.entries(params)) {
    if (key.endsWith('_id') && (typeof value === 'string' || typeof value === 'number')) {
      const rel = key.slice(0, -3);
      result[rel] = { id: { $eq: value } };
    } else if (value !== undefined && value !== null) {
      result[key] = typeof value === 'object' && ('$eq' in value || '$in' in value) ? value : { $eq: value };
    }
  }
  return result;
}

// ── Generic entity factory ────────────────────────────────────────────────────
function createEntityClient(endpoint) {
  return {
    list: async (sort = '-createdAt', limit = 100) => {
      const sortParam = sort.startsWith('-') ? `${sort.substring(1)}:desc` : `${sort}:asc`;
      const q = qs.stringify({ sort: sortParam, populate: 'deep,3', pagination: { limit } }, { encodeValuesOnly: true });
      const res = await api.get(`/${endpoint}?${q}`);
      return flattenResponse(res.data);
    },

    filter: async (params = {}, sort = '-createdAt', limit = 100) => {
      const sortParam = sort.startsWith('-') ? `${sort.substring(1)}:desc` : `${sort}:asc`;
      const q = qs.stringify(
        { filters: buildStrapiFilters(params), populate: 'deep,3', sort: sortParam, pagination: { limit } },
        { encodeValuesOnly: true }
      );
      const res = await api.get(`/${endpoint}?${q}`);
      return flattenResponse(res.data);
    },

    create: async (data) => {
      const res = await api.post(`/${endpoint}`, { data });
      return flattenResponse(res.data);
    },

    bulkCreate: async (items) => {
      return Promise.all(items.map((data) => api.post(`/${endpoint}`, { data }).then((r) => flattenResponse(r.data))));
    },

    update: async (id, data) => {
      const res = await api.put(`/${endpoint}/${id}`, { data });
      return flattenResponse(res.data);
    },

    delete: async (id) => {
      await api.delete(`/${endpoint}/${id}`);
      return { success: true };
    },

    subscribe: (callback) => {
      const eventName = `${endpoint}_event`;
      socket.on(eventName, callback);
      return () => socket.off(eventName, callback);
    },
  };
}

// ── Single-type wrapper (GlobalAdSetting) ────────────────────────────────────
function createSingleTypeClient(endpoint) {
  return {
    filter: async () => {
      try {
        const res = await api.get(`/${endpoint}?populate=*`);
        const item = flattenResponse(res.data);
        return item ? [item] : [];
      } catch {
        return [];
      }
    },
    create: async (data) => {
      const res = await api.put(`/${endpoint}`, { data });
      return flattenResponse(res.data);
    },
    update: async (_id, data) => {
      const res = await api.put(`/${endpoint}`, { data });
      return flattenResponse(res.data);
    },
  };
}

import { Input } from '@/components/ui/input';

import { Sparkles, AlertCircle, Loader2 } from 'lucide-react';

export default function LoginPage() {
  const navigate = useNavigate();
  const [form, setForm] = useState({ email: '', password: '' });
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  // If already logged in, redirect
  useEffect(() => {
    if (base44.auth.isAuthenticated()) {
      navigate(sessionStorage.getItem('loginReturnUrl') || '/', { replace: true });
    }
  }, [navigate]);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');
    setLoading(true);
    try {
      await base44.auth.loginViaEmailPassword(form.email, form.password);
      const returnUrl = sessionStorage.getItem('loginReturnUrl') || '/';
      sessionStorage.removeItem('loginReturnUrl');
      window.location.href = returnUrl;
    } catch (err) {
      setError(err?.response?.data?.error?.message || 'Invalid email or password.');
      setLoading(false);
    }
  };

  return (
    <div className="min-h-screen bg-background flex items-center justify-center p-4">
      <div className="w-full max-w-sm">
        <div className="text-center mb-8">
          <div className="w-12 h-12 rounded-2xl bg-primary/10 flex items-center justify-center mx-auto mb-4">
            <Sparkles className="w-6 h-6 text-primary" />
          </div>
          <h1 className="text-2xl font-heading font-bold text-foreground">Welcome back</h1>
          <p className="text-muted-foreground text-sm mt-1">Sign in to your SmartMenu account</p>
        </div>

        <form onSubmit={handleSubmit} className="space-y-4">
          {error && (
            <div className="flex items-start gap-2 bg-destructive/10 border border-destructive/30 rounded-lg p-3">
              <AlertCircle className="w-4 h-4 text-destructive shrink-0 mt-0.5" />
              <p className="text-destructive text-sm">{error}</p>
            </div>
          )}
          <div className="space-y-1.5">
            <Label>Email</Label>
            <Input
              type="email"
              value={form.email}
              onChange={(e) => setForm((f) => ({ ...f, email: e.target.value }))}
              placeholder="you@example.com"
              required
            />
          </div>
          <div className="space-y-1.5">
            <Label>Password</Label>
            <Input
              type="password"
              value={form.password}
              onChange={(e) => setForm((f) => ({ ...f, password: e.target.value }))}
              placeholder="••••••••"
              required
            />
          </div>
          <Button type="submit" className="w-full h-11 font-bold" disabled={loading}>
            {loading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Signing in…</> : 'Sign In'}
          </Button>
        </form>
      </div>
    </div>
  );
}

function GuestConversionHandler() {
  const { user } = useAuth();
  const [state, setState] = useState('idle');

  useEffect(() => {
    if (!user?.email || state !== 'idle') return;
    const inDemo = isDemoMode();
    const hasIntent = hasConversionIntent();
    if (!inDemo) return;

    if (hasIntent) {
      setState('converting');
      convertDemoToReal(user.email, user.full_name)
        .then(() => { clearConversionIntent(); window.location.href = '/'; })
        .catch(() => { clearDemoMode(); clearConversionIntent(); window.location.href = '/'; });
    } else {
      setState('conflict');
    }
  }, [user?.email]);

  const handleSave = () => {
    setState('converting');
    convertDemoToReal(user.email, user.full_name)
      .then(() => { clearConversionIntent(); window.location.href = '/'; })
      .catch(() => { clearDemoMode(); window.location.href = '/'; });
  };

  const handleExit = () => { clearDemoMode(); clearConversionIntent(); window.location.href = '/'; };

  if (state === 'converting') {
    return (
      <div className="fixed inset-0 flex items-center justify-center bg-[#080808] z-50">
        <div className="text-center">
          <div className="w-10 h-10 border-4 border-amber-400/30 border-t-amber-400 rounded-full animate-spin mx-auto mb-4" />
          <p className="text-white/60 text-sm">Setting up your restaurant…</p>
        </div>
      </div>
    );
  }
  if (state === 'conflict') {
    return <DemoConflictModal onSave={handleSave} onExit={handleExit} saving={false} />;
  }
  return null;
}

   'currentBusinessId', 'guestBusinessId', 'guestDemoSessionId'].forEach((k) => {

  });

}

}

  try { return JSON.parse(localStorage.getItem(DEMO_MENU_KEY)) || demoMenuItems; } catch { return demoMenuItems; }
}

  try { return JSON.parse(localStorage.getItem(DEMO_CATS_KEY)) || demoCategories; } catch { return demoCategories; }
}

export async function convertDemoToReal(userEmail, userFullName, formData = {}) {
  const me = await base44.auth.me();
  const userId = me?.id;

  const menuItems = getDemoMenuItems();
  const categories = getDemoCategories();

  // 1. Create business
  const business = await base44.entities.Business.create({
    business_name: formData.business_name || demoBusiness.business_name,
    business_type: formData.business_type || demoBusiness.business_type,
    country: formData.country || '',
    city: formData.city || '',
    currency: demoBusiness.currency,
    plan_type: 'basic',
    is_active: true,
    is_published: false,
    owner: userId,
  });

  // 2. Create branch
  const branch = await base44.entities.Branch.create({
    branch_name: demoBranch.branch_name,
    business: business.id,
    is_active: true,
  });

  // 3. Create tables
  await base44.entities.Table.bulkCreate(
    demoTables.map((t) => ({
      business: business.id,
      branch: branch.id,
      table_number: t.table_number,
      table_name: t.table_name,
      capacity: t.capacity,
      status: 'available',
    }))
  );

  // 4. Create categories and build ID map
  const catMap = {};
  for (const cat of categories) {
    const created = await base44.entities.MenuCategory.create({
      business: business.id,
      name: cat.name,
      icon: cat.icon,
      sort_order: cat.sort_order,
      is_active: true,
    });
    catMap[cat.id] = created.id;
  }

  // 5. Create menu items
  await Promise.all(
    menuItems.map((item) =>
      base44.entities.MenuItem.create({
        business: business.id,
        menu_category: catMap[item.category_id] || null,
        name: item.name,
        description: item.description,
        price: item.price,
        is_available: item.is_available,
        is_popular: item.is_popular,
      })
    )
  );

  // 6. Create owner employee record
  await base44.entities.Employee.create({
    user: userId,
    business: business.id,
    role: 'owner',
    full_name: userFullName || '',
    is_active: true,
    permissions: {
      manage_menu: true, manage_orders: true, manage_employees: true,
      view_reports: true, manage_tables: true, manage_billing: true,
    },
  });

  clearDemoMode();
  localStorage.setItem('currentBusinessId', String(business.id));
  return business;
}

import { createContext, useContext, useEffect, useState } from 'react';
import { base44 } from '@/api/base44Client';

const PlatformAdminContext = createContext();

export function usePlatformAdmin() {
  const context = useContext(PlatformAdminContext);
  if (!context) throw new Error('usePlatformAdmin must be used within PlatformAdminProvider');
  return context;
}

export function PlatformAdminProvider({ children }) {
  const [platformRole, setPlatformRole] = useState(null);
  const [platformAdminRecord, setPlatformAdminRecord] = useState(null);
  const [isCheckingAccess, setIsCheckingAccess] = useState(true);

  useEffect(() => {
    const check = async () => {
      try {
        if (!base44.auth.isAuthenticated()) { setIsCheckingAccess(false); return; }
        const user = await base44.auth.me();
        if (!user?.email) { setIsCheckingAccess(false); return; }

        const records = await base44.entities.PlatformAdmin.filter({ email: user.email, is_active: true });
        const record = records?.[0];

        if (record) {
          setPlatformRole('platform_master');
          setPlatformAdminRecord(record);
        } else {
          setPlatformRole(null);
          setPlatformAdminRecord(null);
        }
      } catch {
        setPlatformRole(null);
        setPlatformAdminRecord(null);
      } finally {
        setIsCheckingAccess(false);
      }
    };
    check();
  }, []);

  const canAccessPlatformMaster = () => platformRole === 'platform_master' || !!platformAdminRecord?.is_active;

  return (
    <PlatformAdminContext.Provider value={{ platformRole, platformAdminRecord, isCheckingAccess, canAccessPlatformMaster }}>
      {children}
    </PlatformAdminContext.Provider>
  );
}

import { base44 } from '@/api/base44Client';

export async function bootstrapPlatformMaster(user) {
  // With Strapi, platform master is managed via PlatformAdmin records.
  // Auto-bootstrap is only needed if the system is empty and the first user logs in.
  if (!user?.email) return null;
  try {
    const existing = await base44.entities.PlatformAdmin.filter({ email: user.email });
    if (existing?.length > 0) return null; // Already exists
    // Don't auto-create — platform master must be manually assigned via /platform-master setup
    return null;
  } catch {
    return null;
  }
}

export async function canAccessPlatformMaster(user) {
  if (!user?.email) return { canAccess: false, reason: 'No authenticated user' };
  try {
    const records = await base44.entities.PlatformAdmin.filter({ email: user.email, is_active: true });
    if (records?.length > 0) return { canAccess: true, reason: 'PlatformAdmin record found' };
    return { canAccess: false, reason: 'No PlatformAdmin record' };
  } catch {
    return { canAccess: false, reason: 'Error checking access' };
  }
}

import { useState, useEffect } from 'react';
import { base44 } from '@/api/base44Client';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';
import { CheckCircle2, Loader2, AlertCircle, UserCheck } from 'lucide-react';

const ROLE_LABELS = { manager: 'Manager', waiter: 'Waiter' };

export default function AcceptInvitePage() {
  const urlParams = new URLSearchParams(window.location.search);
  const token = urlParams.get('token');

  const [state, setState] = useState('loading');
  const [info, setInfo] = useState(null);
  const [form, setForm] = useState({ password: '', confirmPassword: '' });
  const [error, setError] = useState('');

  useEffect(() => {
    if (!token) { setState('invalid'); return; }
    validateToken();
  }, [token]);

  const validateToken = async () => {
    try {
      const res = await base44.functions.invoke('validateInviteToken', { token });
      const data = res.data;
      if (!data.valid) { setState(data.reason === 'expired' ? 'expired' : 'invalid'); return; }
      setInfo(data);
      if (base44.auth.isAuthenticated()) {
        const me = await base44.auth.me();
        if (me.email === data.invitation.invited_email) {
          setState('completing');
          await completeAcceptance(token);
          return;
        }
      }
      setState('valid');
    } catch {
      setState('invalid');
    }
  };

  const completeAcceptance = async (t) => {
    setState('completing');
    try {
      const res = await base44.functions.invoke('acceptInvite', { token: t });
      const data = res.data;
      if (data.success) {
        localStorage.setItem('currentBusinessId', String(data.business_id));
        setState('success');
        setTimeout(() => { window.location.href = data.role === 'waiter' ? '/waiter' : '/'; }, 2500);
      } else {
        setError(data.error || 'Failed to accept invitation.');
        setState('error');
      }
    } catch (err) {
      setError(err.message);
      setState('error');
    }
  };

  const handleRegister = async (e) => {
    e.preventDefault();
    setError('');
    if (form.password.length < 6) { setError('Password must be at least 6 characters.'); return; }
    if (form.password !== form.confirmPassword) { setError('Passwords do not match.'); return; }
    setState('registering');
    try {
      await base44.auth.register({ email: info.invitation.invited_email, password: form.password });
      await base44.auth.updateMe({ full_name: info.invitation.invited_name });
      await completeAcceptance(token);
    } catch (err) {
      setError(err?.response?.data?.error?.message || 'Registration failed. You may already have an account — try logging in first.');
      setState('valid');
    }
  };

  return (
    <div className="min-h-screen bg-[#080808] flex items-center justify-center p-4">
      <div className="w-full max-w-md bg-card border border-border rounded-2xl overflow-hidden shadow-2xl">
        <div className="bg-amber-400/10 border-b border-amber-400/20 px-6 py-4 flex items-center gap-3">
          <div className="w-8 h-8 rounded-lg bg-amber-400/15 flex items-center justify-center">
            <UserCheck className="w-4 h-4 text-amber-400" />
          </div>
          <span className="text-amber-400 font-heading font-bold">Employee Invitation</span>
        </div>

        <div className="p-6">
          {state === 'loading' && <div className="py-8 text-center space-y-3"><Loader2 className="w-8 h-8 text-amber-400 animate-spin mx-auto" /><p className="text-white/50 text-sm">Validating your invitation…</p></div>}

          {state === 'invalid' && <div className="py-8 text-center space-y-4"><AlertCircle className="w-10 h-10 text-destructive mx-auto" /><div><h2 className="text-white font-bold text-lg">Invalid Invitation Link</h2><p className="text-white/50 text-sm mt-1">This invitation link is invalid or has been cancelled.</p></div></div>}

          {state === 'expired' && <div className="py-8 text-center space-y-4"><AlertCircle className="w-10 h-10 text-warning mx-auto" /><div><h2 className="text-white font-bold text-lg">Invitation Expired</h2><p className="text-white/50 text-sm mt-1">Ask your manager to resend the invitation.</p></div></div>}

          {(state === 'completing' || state === 'registering') && <div className="py-8 text-center space-y-3"><Loader2 className="w-8 h-8 text-amber-400 animate-spin mx-auto" /><p className="text-white/50 text-sm">Setting up your account…</p></div>}

          {state === 'success' && (
            <div className="py-8 text-center space-y-4">
              <div className="w-16 h-16 rounded-full bg-emerald-500/15 flex items-center justify-center mx-auto"><CheckCircle2 className="w-8 h-8 text-emerald-400" /></div>
              <div><h2 className="text-white font-bold text-xl">Welcome to the team!</h2><p className="text-white/50 text-sm mt-1">Redirecting you to your dashboard…</p></div>
            </div>
          )}

          {state === 'error' && (
            <div className="py-8 text-center space-y-4">
              <AlertCircle className="w-10 h-10 text-destructive mx-auto" />
              <div><h2 className="text-white font-bold text-lg">Something went wrong</h2><p className="text-white/50 text-sm mt-1">{error}</p></div>
              <Button variant="outline" onClick={validateToken}>Try Again</Button>
            </div>
          )}

          {state === 'valid' && info && (
            <div className="space-y-5">
              <div className="bg-secondary/50 rounded-xl p-4 space-y-2">
                <div className="flex justify-between text-sm"><span className="text-white/40">Business</span><span className="text-amber-400 font-semibold">{info.business?.business_name}</span></div>
                <div className="flex justify-between text-sm"><span className="text-white/40">Invited as</span><span className="text-white font-medium capitalize">{ROLE_LABELS[info.invitation.role]}</span></div>
                <div className="flex justify-between text-sm"><span className="text-white/40">Email</span><span className="text-white/70">{info.invitation.invited_email}</span></div>
              </div>
              {error && (
                <div className="flex items-start gap-2 bg-destructive/10 border border-destructive/30 rounded-lg p-3">
                  <AlertCircle className="w-4 h-4 text-destructive shrink-0 mt-0.5" />
                  <p className="text-destructive text-sm">{error}</p>
                </div>
              )}
              <form onSubmit={handleRegister} className="space-y-4">
                <div className="space-y-1"><Label className="text-white/60 text-xs">Create Password *</Label><Input type="password" placeholder="Min. 6 characters" value={form.password} onChange={(e) => setForm((f) => ({ ...f, password: e.target.value }))} required minLength={6} /></div>
                <div className="space-y-1"><Label className="text-white/60 text-xs">Confirm Password *</Label><Input type="password" placeholder="Repeat password" value={form.confirmPassword} onChange={(e) => setForm((f) => ({ ...f, confirmPassword: e.target.value }))} required /></div>
                <Button type="submit" className="w-full bg-amber-400 text-black hover:bg-amber-300 font-bold h-11">Create Account & Join Team</Button>
              </form>
            </div>
          )}

            <div className="flex gap-3"><Button variant="outline" className="flex-1" onClick={() => setStep(2)} disabled={saving}>Back</Button>
              <Button className="flex-1" onClick={handleSubmit} disabled={saving}>
                {saving ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Creating...</> : 'Create Business'}

export default function ReportsPage() {
  const { currentBusinessId } = useBusinessContext();

  const { data: orders = [], isLoading } = useQuery({
    queryKey: ['allOrders', currentBusinessId],
    queryFn: () => base44.entities.Order.filter({ business_id: currentBusinessId }, '-createdAt', 500),
    enabled: !!currentBusinessId,
  });

  const completedOrders = orders.filter(o => o.status === 'completed' || o.status === 'served');
  const totalSales = completedOrders.reduce((s, o) => s + (o.total || 0), 0);
  const avgOrder = completedOrders.length ? totalSales / completedOrders.length : 0;

  // Sales by day (last 7 days) — Strapi uses createdAt
  const last7 = [];
  for (let i = 6; i >= 0; i--) {
    const d = new Date();
    d.setDate(d.getDate() - i);
    const dateStr = d.toDateString();
    const dayOrders = completedOrders.filter(o => new Date(o.createdAt).toDateString() === dateStr);
    last7.push({
      day: d.toLocaleDateString('en', { weekday: 'short' }),
      sales: dayOrders.reduce((s, o) => s + (o.total || 0), 0),
      orders: dayOrders.length,
    });
  }

  const itemCounts = {};
  completedOrders.forEach(o => {
    o.items?.forEach(item => {
      itemCounts[item.name] = (itemCounts[item.name] || 0) + item.quantity;
    });
  });
  const topItems = Object.entries(itemCounts).sort((a, b) => b[1] - a[1]).slice(0, 5).map(([name, value]) => ({ name, value }));

  if (isLoading) return <div className="space-y-4">{[1,2,3].map(i => <Skeleton key={i} className="h-32" />)}</div>;

  return (
    <div className="space-y-6">

        <StatCard title="Total Revenue" value={`$${totalSales.toFixed(2)}`} icon={DollarSign} />
        <StatCard title="Total Orders" value={completedOrders.length} icon={ShoppingCart} />
        <StatCard title="Avg Order" value={`$${avgOrder.toFixed(2)}`} icon={TrendingUp} />
        <StatCard title="Top Items" value={topItems.length} icon={Award} />
      </div>

      <div className="grid md:grid-cols-2 gap-6">
        <Card className="p-6">
          <h3 className="font-heading font-semibold mb-4">Sales (Last 7 Days)</h3>
          <ResponsiveContainer width="100%" height={250}>
            <BarChart data={last7}>
              <CartesianGrid strokeDasharray="3 3" stroke="hsl(var(--border))" />
              <XAxis dataKey="day" stroke="hsl(var(--muted-foreground))" fontSize={12} />
              <YAxis stroke="hsl(var(--muted-foreground))" fontSize={12} />
              <Tooltip contentStyle={{ background: 'hsl(var(--card))', border: '1px solid hsl(var(--border))', borderRadius: '8px' }} />
              <Bar dataKey="sales" fill="hsl(211,83%,43%)" radius={[4, 4, 0, 0]} />

          </ResponsiveContainer>
        </Card>

        <Card className="p-6">

          {topItems.length === 0 ? (

          ) : (

              <ResponsiveContainer width="50%" height={200}>

                  <Pie data={topItems} dataKey="value" cx="50%" cy="50%" outerRadius={80} innerRadius={50}>

        </div>
        <div><Label>Service Charge (%)</Label><Input type="number" value={form.service_charge_percent || 0} onChange={e => setForm(p => ({ ...p, service_charge_percent: parseFloat(e.target.value) || 0 }))} className="mt-1" /></div>

      <Card className="p-6">
        <h2 className="font-semibold mb-4">Subscription Plan</h2>
        <div className="grid grid-cols-2 gap-4">

            <button key={plan}
              onClick={() => plan === 'premium' ? guardAction(() => setForm(p => ({ ...p, plan_type: plan }))) : setForm(p => ({ ...p, plan_type: plan }))}
              className={`p-5 rounded-lg border text-left transition-all ${form.plan_type === plan ? 'border-primary bg-primary/10' : 'border-border hover:border-primary/30'}`}
            >
              {plan === 'premium' ? (isDemo ? <Lock className="w-5 h-5 text-primary mb-2" /> : <Crown className="w-5 h-5 text-primary mb-2" />) : <Sparkles className="w-5 h-5 text-muted-foreground mb-2" />}
              <p className="font-bold capitalize">{plan}</p>
              <p className="text-xs text-muted-foreground mt-1">{plan === 'basic' ? 'QR Menu + Ads' : 'AI Features + No Ads'}</p>
            </button>
          ))}
        </div>
      </Card>

      <Card className="p-6 space-y-4 border-destructive/20 bg-destructive/5">
        <h2 className="font-semibold text-sm text-destructive">Account Management</h2>
        <p className="text-sm text-muted-foreground">Permanently delete your account and all associated data. This cannot be undone.</p>
        <Button variant="destructive" className="w-full sm:w-auto" onClick={() => setShowDeleteModal(true)}>
          <Trash2 className="w-4 h-4 mr-2" /> Delete Account
        </Button>
      </Card>

      {showModal && <DemoLockModal onClose={closeModal} />}
      <DeleteAccountModal open={showDeleteModal} onOpenChange={setShowDeleteModal} userEmail={user?.email} />

      <Button size="lg" className="w-full" disabled={update.isPending} onClick={() => guardAction(() => update.mutate(form))}>
        {isDemo && <Lock className="w-4 h-4 mr-2 opacity-70" />}
        <Save className="w-4 h-4 mr-2" /> {update.isPending ? 'Saving…' : 'Save Settings'}
      </Button>
    </div>
  );
}

// FILE: src/pages/AdsManagement.jsx
import { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';
import { useBusinessContext } from '@/hooks/useBusinessContext';
import { Button } from '@/components/ui/button';
import { Switch } from '@/components/ui/switch';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { Megaphone, Plus, Lock, Crown, Info } from 'lucide-react';
import { useGuestGuard } from '@/hooks/useGuestGuard';
import DemoLockModal from '@/components/demo/DemoLockModal';
import AdCard from '@/components/ads/AdCard';
import CampaignFormDialog from '@/components/ads/CampaignFormDialog';

const adModeLabels = {
  platform_ads: 'Platform Ads (with revenue share)',
  business_ads_only: 'My Promotions Only',
  no_ads: 'No Ads',
  mixed_ads: 'Mixed Ads',
  google_adsense: 'Google AdSense',
};

export default function AdsManagement() {
  const { currentBusinessId, isPremium, business } = useBusinessContext();
  const queryClient = useQueryClient();
  const { isDemo, showModal, closeModal, guardAction } = useGuestGuard();
  const [dialog, setDialog] = useState(false);
  const [editing, setEditing] = useState(null);

  const { data: campaigns = [] } = useQuery({
    queryKey: ['businessCampaigns', currentBusinessId],
    queryFn: () => base44.entities.AdCampaign.filter({ business_id: currentBusinessId }),
    enabled: !!currentBusinessId,
  });

  const { data: adSettingsArr = [] } = useQuery({
    queryKey: ['bizAdSettings', currentBusinessId],
    queryFn: () => base44.entities.BusinessAdSettings.filter({ business_id: currentBusinessId }),
    enabled: !!currentBusinessId,
  });

  const adSettings = adSettingsArr[0];

  const updateAdMode = useMutation({
    mutationFn: async (ad_mode) => {
      if (adSettings) {
        await base44.entities.BusinessAdSettings.update(adSettings.id, { ad_mode });
      } else {
        await base44.entities.BusinessAdSettings.create({
          business: currentBusinessId,
          ad_mode,
        });
      }
    },
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['bizAdSettings'] }),
  });

  const toggleCampaign = useMutation({
    mutationFn: ({ campaign, status }) => base44.entities.AdCampaign.update(campaign.id, { status }),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['businessCampaigns'] }),
  });

  const deleteCampaign = useMutation({
    mutationFn: (id) => base44.entities.AdCampaign.delete(id),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['businessCampaigns'] }),
  });

  const openNew = () => { setEditing(null); setDialog(true); };
  const openEdit = (c) => { setEditing(c); setDialog(true); };

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-heading font-bold">Advertisements</h1>
          <p className="text-muted-foreground text-sm mt-0.5">Manage your promotions and ad preferences</p>
        </div>
        <Button onClick={() => guardAction(openNew)} className="gap-1.5">
          {isDemo && <Lock className="w-3.5 h-3.5 opacity-70" />}
          <Plus className="w-4 h-4" /> New Promotion
        </Button>
      </div>

      <div className={`rounded-2xl border p-4 ${isPremium ? 'border-primary/20 bg-primary/5' : 'border-border bg-card'}`}>
        <div className="flex items-start gap-3">
          {isPremium ? <Crown className="w-5 h-5 text-primary shrink-0 mt-0.5" /> : <Info className="w-5 h-5 text-muted-foreground shrink-0 mt-0.5" />}
          <div className="flex-1">
            {isPremium ? (
              <>
                <p className="font-semibold text-primary text-sm">Premium Plan — You control your ads</p>
                <p className="text-xs text-muted-foreground mt-1">Choose how ads appear on your menu:</p>
                <div className="mt-3">
                  <Select value={adSettings?.ad_mode || 'no_ads'} onValueChange={(v) => updateAdMode.mutate(v)}>
                    <SelectTrigger className="bg-muted border-border w-72"><SelectValue /></SelectTrigger>
                    <SelectContent>
                      {Object.entries(adModeLabels).map(([k, l]) => <SelectItem key={k} value={k}>{l}</SelectItem>)}
                    </SelectContent>
                  </Select>
                </div>
              </>
            ) : (
              <>
                <p className="font-semibold text-foreground text-sm">Basic Plan — Platform ads are active</p>
                <p className="text-xs text-muted-foreground mt-1">Your menu shows platform ads by default. Upgrade to Premium to control or remove ads.</p>
                <Button variant="outline" size="sm" className="mt-3 border-primary/30 text-primary hover:bg-primary/10 h-8 text-xs">
                  <Crown className="w-3.5 h-3.5 mr-1" /> Upgrade to Premium
                </Button>
              </>
            )}
          </div>
        </div>
      </div>

      <div>
        <h2 className="font-semibold text-foreground mb-3">My Promotions</h2>
        {campaigns.length === 0 && (
          <div className="border border-dashed border-border rounded-2xl py-10 text-center">
            <Megaphone className="w-8 h-8 text-muted-foreground mx-auto mb-2" />
            <p className="text-muted-foreground text-sm">No promotions yet.</p>
          </div>
        )}
        <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
          {campaigns.map(c => (
            <AdCard
              key={c.id}
              campaign={c}
              onEdit={openEdit}
              onDelete={(id) => deleteCampaign.mutate(id)}
              onToggle={(camp, status) => toggleCampaign.mutate({ campaign: camp, status })}
            />
          ))}
        </div>
      </div>

      {showModal && <DemoLockModal onClose={closeModal} />}
      <CampaignFormDialog
        open={dialog}
        onClose={() => setDialog(false)}
        campaign={editing}
        businessId={currentBusinessId}
        onSaved={() => queryClient.invalidateQueries({ queryKey: ['businessCampaigns'] })}
      />
    </div>
  );
}
